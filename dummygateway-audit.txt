const PAD = 3;

// ---- payment modes ----
const PAYMENT_MODES = ["BANK"];

function pick(arr) {
  return arr[Math.floor(Math.random() * arr.length)];
}

// ---- read last used counter ----
let last = 0;
if (fs.existsSync(COUNTER_FILE)) {
  const raw = fs.readFileSync(COUNTER_FILE, "utf8").trim();
  const parsed = parseInt(raw.replace(/\D/g, ""), 10);
  if (Number.isFinite(parsed)) last = parsed;
}

const BATCH = 500; // change per run
const start = last + 1;
const end = last + BATCH;

const rows = [
  "amount,merchantReference,paymentMode,beneficiaryName,beneficiaryAccount,beneficiaryIfsc,beneficiaryBank,email,mobile",
];

for (let i = start; i <= end; i++) {
  const ref = `${PREFIX}${String(i).padStart(PAD, "0")}`;
  const mode = pick(PAYMENT_MODES);
  rows.push(
    `10,${ref},${mode},Rahul Sharma,1000000001,HDFC0001234,HDFC Bank,rahul@example.com,9876543210`,
  );
}

const outFile = `payouts-${start}-${end}.csv`;

 * Called by PayoutMock itself (or you, from Postman) to simulate a new dispute.
 *
 * Body: { reference_id, amount, currency, reason, reason_description }
 */
router.post("/create", async (req, res) => {
  try {
    const { reference_id, amount, currency, reason, reason_description } =
      req.body || {};

    if (!reference_id) {
      return res.status(400).json({ success: false, message: "reference_id is required" });
    }

    const payin = payinStore.getPayin(reference_id);
    const cbAmount = Number(amount || payin?.amount || 0);
    if (!cbAmount || cbAmount <= 0) {
      return res.status(400).json({ success: false, message: "amount is required" });
    }

    const gatewayChargebackId = `DUMMY_CB_${Date.now()}_${uuid().slice(0, 6)}`;

    chargebackStore.createChargeback({
      gatewayChargebackId,
      referenceId: reference_id,
      amount: cbAmount,
      currency: currency || "INR",
      reason: reason || "customer_dispute",
      reasonDescription: reason_description || null,
      behaviour: "CHARGEBACK_DISPUTED",
    });

    // Fire CHARGEBACK_CREATED webhook immediately
    const delayMs = Number(process.env.AUTO_CALLBACK_DELAY_MS || 5000);
    const auto = String(process.env.AUTO_CALLBACK_ENABLED || "true") === "true";
    if (auto) {
      setTimeout(() => {
        dispatcher
          .dispatchChargebackCreated(gatewayChargebackId, {
            reason: reason || "customer_dispute",
            amount: cbAmount,
          })
          .catch((e) =>
            console.error("[PayoutMock] chargeback create callback error:", e.message)
          );
      }, delayMs);
    }

    return res.json({
      success: true,
      data: {
        gateway_chargeback_id: gatewayChargebackId,
        reference_id,
        amount: cbAmount,
        currency: currency || "INR",
        status: "DISPUTED",
        reason: reason || "customer_dispute",
        message: auto
          ? `CHARGEBACK_CREATED webhook in ${delayMs}ms`

  try {
    const {
      reference_id,
      amount,
      currency,
      payment_mode,
      customer_name,
      customer_email,
      customer_phone,
      return_url,
    } = req.body || {};

    if (!reference_id) {
      return res
        .status(400)
        .json({ success: false, message: "reference_id is required" });
    }
    if (!amount || Number(amount) <= 0) {
      return res
        .status(400)
        .json({ success: false, message: "amount must be > 0" });
    }

    // Idempotent re-issue
    const existing = payinStore.getPayin(reference_id);
    if (existing) {
      return res.json({

        idempotent: true,
        data: {
          reference_id: existing.referenceId,

          amount: existing.amount,
          checkout_url: `${process.env.PAYOUT_MOCK_URL || "http://localhost:4100"}/pay/${encodeURIComponent(existing.referenceId)}`,
        },
      });
    }

    // Behaviour: first digit of reference_id or default
    const first = String(reference_id).charAt(0);
    const behaviour =
      first === "2" ? "PAYIN_FAIL" : "PAYIN_SUCCESS_AFTER_DELAY";

    const transferId = `DUMMY_PAYIN_${Date.now()}_${uuid().slice(0, 6)}`;
    const mockBase =
      process.env.PAYOUT_MOCK_URL ||
      `http://localhost:${process.env.PAYOUT_MOCK_PORT || 4100}`;
    const checkoutUrl = `${mockBase}/pay/${encodeURIComponent(reference_id)}`;

    payinStore.createPayin({
      referenceId: reference_id,
      transferId,
      amount: Number(amount),
      currency: currency || "INR",
      paymentMode: payment_mode || "PG",
      customerName: customer_name,
      customerEmail: customer_email,
      customerPhone: customer_phone,
      behaviour,
    });

    // Read once â€” used by both success and failure paths
    const delayMs = Number(process.env.AUTO_CALLBACK_DELAY_MS || 0);
    const auto = String(process.env.AUTO_CALLBACK_ENABLED || "true") === "true";

    // -------- Immediate business failure --------
    if (behaviour === "PAYIN_FAIL") {
      payinStore.updatePayin(reference_id, {
        status: "FAILED",
        failureReason: "Payment declined by bank",
        failureCode: "PAYMENT_FAILED",
      });

      if (auto) {
        const fireFail = () =>
          dispatcher
            .dispatchPayinFailed(reference_id, {
              reason: "Payment declined by bank",
              code: "PAYMENT_FAILED",
            })
            .catch((e) =>
              console.error("[PayoutMock] payin fail callback error:", e.message),
            );

        else fireFail();

      return res.status(200).json({
        success: false,
        message: "Payment declined by bank",
        failure_code: "PAYMENT_FAILED",
        reference_id: reference_id,
        gateway_reference: transferId,
        checkout_url: checkoutUrl,
        return_url: return_url || null,
      });
    }

    // -------- Success path: fire callback (inline when delay=0) --------
    if (auto) {
      const fireSuccess = () =>

          .catch((e) =>
            console.error(
              "[PayoutMock] payin success callback error:",
              e.message,
            ),
          );

      else fireSuccess();
    }

    return res.json({

      data: {
        reference_id: reference_id,
        gateway_reference: transferId,
        status: "PROCESSING",
        amount: Number(amount),
        currency: currency || "INR",
        checkout_url: checkoutUrl,
        return_url: return_url || null,

          : "Awaiting manual webhook via POST /webhook/fire",
      },
    });

    console.error("[PayoutMock] /payin/create error:", error);

const express = require("express");
const { v4: uuid } = require("uuid");
const bankAccounts = require("../services/bank-account.service");
const payoutStore = require("../services/payout-store.service");
const dispatcher = require("../services/webhook-dispatcher.service");

const router = express.Router();

/**
 * POST /payouts
 * Called by fintech-backend's DummyPayoutIntegrationService.
 *
 * Behaviour: PENDING immediately + auto webhook fires right away
 * (controlled by AUTO_CALLBACK_ENABLED / AUTO_CALLBACK_DELAY_MS).
 */
router.post("/", async (req, res) => {
  try {
    const {
      reference_id,
      transfer_amount,
      beneficiary_account,
      beneficiary_ifsc,
      beneficiary_name,
      idempotency_key,
    } = req.body || {};

    if (!reference_id) {
      return res
        .status(400)
        .json({ success: false, message: "reference_id is required" });
    }
    if (!transfer_amount || Number(transfer_amount) <= 0) {
      return res
        .status(400)
        .json({ success: false, message: "transfer_amount must be > 0" });
    }

    // Idempotency: if this reference is already known, return same result
    const existing = payoutStore.getPayout(reference_id);
    if (existing) {
      return res.json({
        success: true,
        idempotent: true,
        data: {
          reference_id: existing.referenceId,
          transfer_id: existing.transferId,
          transfer_status:
            existing.status === "COMPLETED"
              ? "SUCCESS"
              : existing.status === "FAILED"
                ? "FAILED"
                : "PENDING",
          transfer_amount: existing.amount,
          bank_reference: existing.utr,
          failure_reason: existing.failureReason,
          failure_code: existing.failureCode,
        },
      });
    }

    const transferId = `DUMMY_${Date.now()}_${uuid().slice(0, 6)}`;
    const account = bankAccounts.getAccount(beneficiary_account);

    // ---------- Decide behaviour ----------
    let behaviour = account?.behaviour || null;
    if (!behaviour) {
      const first = String(beneficiary_account || "").charAt(0);
      behaviour =
        first === "2"
          ? "BUSINESS_FAIL"
          : first === "5"
            ? "TIMEOUT"
            : first === "6"
              ? "RETRY_503"
              : first === "7"
                ? "NEVER_CALLBACK"
                : first === "4"
                  ? "PENDING_THEN_FAIL"
                  : "PENDING_THEN_SUCCESS";
    }

    // Save record
    payoutStore.createPayout({
      referenceId: reference_id,
      transferId,
      amount: Number(transfer_amount),
      accountNumber: beneficiary_account,
      beneficiaryName: beneficiary_name,
      ifsc: beneficiary_ifsc,
      bank: account?.bank || "Unknown",
      behaviour,
    });

    const fail = (reason, code, extra = {}) => {
      payoutStore.updatePayout(reference_id, {
        status: "FAILED",
        failureReason: reason,
        failureCode: code,
      });
      const data = {
        transfer_id: transferId,
        transfer_status: "FAILED",
        transfer_amount: Number(transfer_amount),
        reference_id: reference_id,
        failure_reason: reason,
        failure_code: code,
        ...extra,
      };
      return res.json({ success: true, data });
    };

    // ---------- Validation ----------
    const ifsc = beneficiary_ifsc ? String(beneficiary_ifsc).toUpperCase() : "";
    if (!/^[A-Z]{4}0[A-Z0-9]{6}$/.test(ifsc)) {
      return fail(`Invalid IFSC: ${beneficiary_ifsc}`, "BENEFICIARY_INVALID");
    }
    if (!account) {

const axios = require("axios");
const crypto = require("crypto");
const bankAccountService = require("./bank-account.service");

const BACKEND_URL =
  process.env.BACKEND_WEBHOOK_URL || "http://localhost:4000/api/webhooks/dummy";
const SECRET = process.env.DUMMY_WEBHOOK_SECRET || "dummy-webhook-secret";

/**
 * Build the webhook payload in the shape dummy.webhook.js expects.
 */
function buildPayload({
  merchantReference,
  transferId,
  status,
  utr,
  amount,
  failureReason,
  failureCode,
}) {
  const normalised = String(status || "COMPLETED").toUpperCase();

  const transferStatus =
    normalised === "COMPLETED"
      ? "SUCCESS"
      : normalised === "FAILED"
        ? "FAILED"
        : normalised === "PENDING"
          ? "PENDING"
          : normalised;

  const event =
    normalised === "COMPLETED"
      ? "PAYOUT_SUCCESS"
      : normalised === "FAILED"
        ? "PAYOUT_FAILED"
        : "PAYOUT_PENDING";

  return {
    event,
    type: event,
    reference_id: merchantReference || null,
    transfer_id: transferId || `DUMMY_${Date.now()}`,
    transfer_status: transferStatus,
    transfer_amount: amount != null ? Number(amount) : null,
    bank_reference: utr || null,
    failure_reason: failureReason || null,
    failure_code: failureCode || null,
    fired_at: new Date().toISOString(),
  };
}

function sign(payload) {
  return crypto
    .createHmac("sha256", SECRET)
    .update(JSON.stringify(payload))
    .digest("hex");
}

/**
 * Fire one webhook (unsigned but headers include signature).
 */
async function fireOnce(payload) {
  const signature = sign(payload);

  const started = Date.now();
  try {
    const response = await axios.post(BACKEND_URL, payload, {
      headers: {
        "Content-Type": "application/json",
        "x-dummy-signature": signature,
        "x-webhook-signature": signature,
      },
      timeout: 10000,
      validateStatus: () => true,
    });

    const result = {
      ok: response.status >= 200 && response.status < 300,
      httpStatus: response.status,
      responseBody: response.data,
      latencyMs: Date.now() - started,
      signed: signature,
    };
    bankAccountService.logWebhook({ ...payload, result });
    return result;
  } catch (error) {
    const result = {
      ok: false,
      httpStatus: 0,
      responseBody: error.message,
      latencyMs: Date.now() - started,
      signed: signature,
    };
    bankAccountService.logWebhook({ ...payload, result });
    throw Object.assign(error, { result });
  }
}

/**
 * Fire webhook. Supports:
 *  - duplicate:      fires the same eventId twice (2s apart)
 *  - delaySeconds:   fires after N seconds (fire-and-forget)
 */
async function fire({
  merchantReference,
  transferId,
  status = "COMPLETED",
  utr,
  amount,
  failureReason,
  failureCode,
  delaySeconds = 0,
  duplicate = false,
}) {
  if (!merchantReference) {
    throw new Error("merchantReference is required");
  }

  const payload = buildPayload({
    merchantReference,
    transferId,
    status,
    utr:
      utr ||
      (String(status).toUpperCase() === "COMPLETED"
        ? `UTR${Date.now()}`
        : null),
    amount,
    failureReason,
    failureCode,
  });

  // Delayed fire-and-forget
  if (delaySeconds && Number(delaySeconds) > 0) {
    const ms = Number(delaySeconds) * 1000;
    setTimeout(() => {
      fireOnce(payload).catch((err) =>
        console.error("[PayoutMock] Delayed fire failed:", err.message),
      );
      if (duplicate) {
        setTimeout(() => {
          fireOnce(payload).catch((err) =>
            console.error(
              "[PayoutMock] Delayed duplicate fire failed:",
              err.message,
            ),
          );
        }, 2000);
      }
    }, ms);

    return {
      scheduled: true,
      delaySeconds: Number(delaySeconds),
      duplicate: Boolean(duplicate),
      payload,
    };
  }

  // Immediate
  const result = await fireOnce(payload);

  // Duplicate test
  let duplicateResult = null;
  if (duplicate) {
    await new Promise((r) => setTimeout(r, 2000));
    duplicateResult = await fireOnce(payload);
  }

  return {
    scheduled: false,
    payload,
    result,
    duplicate: duplicate ? duplicateResult : null,
  };
}

module.exports = { fire, buildPayload, sign };
